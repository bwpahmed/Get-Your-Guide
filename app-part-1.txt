const $ = (selector, root = document) => root.querySelector(selector);
const $$ = (selector, root = document) => [...root.querySelectorAll(selector)];
const esc = (value = '') => String(value).replace(/[&<>'"]/g, (char) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', "'": '&#39;', '"': '&quot;' })[char]);
const money = (value) => `${esc(data.settings.currency)} ${Number(value || 0).toLocaleString()}`;

const visibleCategories = () => data.categories.filter((item) => item.visible).sort((a, b) => a.order - b.order);
const visiblePackages = () => data.packages.filter((item) => item.visible).sort((a, b) => a.order - b.order);
const headerVisible = () => data.sections.find((section) => section.type === 'header')?.visible !== false;
const footerVisible = () => data.sections.find((section) => section.type === 'footer')?.visible !== false;

let selectedCategoryId = visibleCategories()[0]?.id || 'canal';
let selectedPackageId = visiblePackages().find((item) => item.categoryId === selectedCategoryId)?.id || visiblePackages()[0]?.id;
let activeView = document.body.dataset.page === 'package' ? 'experience' : 'overview';

const requestedSlug = new URLSearchParams(location.search).get('slug');
if (requestedSlug) {
  const requested = visiblePackages().find((item) => item.slug === requestedSlug);
  if (requested) {
    selectedPackageId = requested.id;
    selectedCategoryId = requested.categoryId;
    activeView = 'experience';
  }
}

function applyTheme() {
  document.documentElement.style.setProperty('--blue', data.settings.primaryColor === '#0b1f33' ? '#2563eb' : (data.settings.primaryColor || '#2563eb')); 
  document.documentElement.style.setProperty('--accent', data.settings.accentColor || '#d6a756');
  document.title = data.settings.brandName || 'Get Your Guide Dubai';
}

function selectedPackage() {
  return visiblePackages().find((item) => item.id === selectedPackageId) || visiblePackages()[0];
}

function selectedCategory() {
  return visibleCategories().find((item) => item.id === selectedCategoryId) || visibleCategories()[0];
}
